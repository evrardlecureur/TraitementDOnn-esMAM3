\begin{frame}{\faLeaf\quad Slide 1 — Business Goal}
\vspace{-0.3cm}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
  \textbf{\textcolor{coffeebrown}{Le dataset CQI Arabica}}\\[0.3cm]
  Le Coffee Quality Institute (CQI) évalue des cafés arabica
  selon un protocole standardisé de \textit{cupping} — 8 critères
  sensoriels notés de 0 à 10 par des juges certifiés Q-Grader.\\[0.4cm]

  \begin{statbox}
    \begin{center}
      \textbf{207} échantillons \quad|\quad \textbf{23} pays \quad|\quad \textbf{41} variables\\
      Score moyen : \textbf{83,7 / 100} \quad|\quad Max : \textbf{89,3 pts}
    \end{center}
  \end{statbox}
  \vspace{0.4cm}

  \textbf{\textcolor{coffeebrown}{Pourquoi ce dataset ?}}
  \begin{itemize}\small
    \item Notes \textbf{standardisées} par des experts certifiés
    \item Grande diversité : pays, variétés, traitements, altitudes
    \item Idéal pour un pipeline complet : wrangling $\to$ régression
  \end{itemize}
\end{column}
\begin{column}{0.45\textwidth}
  \begin{insightbox}[Question de recherche]
    \small Quels facteurs — sensoriels, géographiques, variétaux —
    permettent de \textbf{prédire et d'expliquer} la qualité
    d'un café arabica certifié CQI ?
  \end{insightbox}

% SLIDE 3 — HANDCRAFTED FEATURES
% ============================================================
\begin{frame}{\faWrench\quad Slide 3 — Features construites}
\vspace{-0.3cm}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
  \textbf{\textcolor{coffeebrown}{3 nouvelles features créées}}\\[0.3cm]

  \begin{block}{1. Indice sensoriel composite}
    \small
    $\text{Sensory} = \dfrac{\text{Flavor} + \text{Aftertaste} + \text{Balance}}{3}$\\
    \textit{Concentre les 3 critères les plus prédictifs ($r > 0.93$)}
  \end{block}
  \vspace{0.3cm}

  \begin{block}{2. Pénalité de défauts (inspirée Lecture 4)}
    \small
    $\text{Penalty} = 4 \times \text{Cat1} + \text{Cat2} + \text{Quakers}$\\
    \textit{Corrélation avec le score : $r = -0.38$}
  \end{block}
  \vspace{0.3cm}

  \begin{block}{3. Altitude en 6 bandes ordinales}
    \small
    $<$1000m / 1000--1200 / 1200--1400 / 1400--1600 / 1600--1800 / $>$1800m\\
    \textit{One-hot encoding — effet monotone visible à haute altitude}
  \end{block}
\end{column}
\begin{column}{0.46\textwidth}
  \centering
  \textbf{\textcolor{coffeebrown}{Justification des features}}\\[0.2cm]
  \includegraphics[width=\textwidth]{fig7_defects.png}
  {\tiny\textcolor{gray}{Gauche : défauts Cat.2 vs score ($r=-0.31$). Droite : Quakers vs score ($r=-0.32$)}}\\[0.3cm]
  \begin{insightbox}[Enseignement clé]
    \small La pureté du grain (peu de défauts) est aussi
    importante que ses arômes pour définir un bon café.
  \end{insightbox}
\end{column}
\end{columns}
\end{frame}

% SLIDE 5 — CONCLUSION

\begin{frame}{\faAward\quad Slide 5 — Conclusion}
\vspace{-0.2cm}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
  \textbf{\textcolor{coffeebrown}{Ce que les données nous ont appris}}\\[0.3cm]
  \begin{enumerate}\small
    \item[\textcolor{coffeegold}{\faCheckCircle}]
      \textbf{Flavor \& Aftertaste} — critères les plus déterminants :
      la persistance aromatique est la signature de l'excellence.
    \item[\textcolor{coffeegold}{\faCheckCircle}]
      \textbf{Éthiopie et Tanzanie} dominent le classement mondial
      (score médian $>$ 84,7 pts).
    \item[\textcolor{coffeegold}{\faCheckCircle}]
      La variété \textbf{Gesha} se détache nettement (85,4 pts moyen).
    \item[\textcolor{coffeegold}{\faCheckCircle}]
      Haute altitude ($>$ 1800 m) $\Rightarrow$ +1,3 pt en moyenne.
    \item[\textcolor{coffeegold}{\faCheckCircle}]
      \textbf{Défauts} (Quakers, Cat.2) pénalisent le score
      de façon mesurable ($r \approx -0.32$).
  \end{enumerate}\\[0.3cm]

  \begin{statbox}
    \small Un modèle logistique sur 6 critères sensoriels
    prédit les cafés d'exception avec \textbf{90\,\% de précision}.
  \end{statbox}
\end{column}
\begin{column}{0.42\textwidth}
  \centering
  \includegraphics[width=0.95\textwidth]{fig8_heatmap_country_method.png}
  {\tiny\textcolor{gray}{Score moyen par pays × méthode de traitement}}\\[0.3cm]
  \textbf{\textcolor{coffeebrown}{Axes d'amélioration}}
  \begin{itemize}\small
    \item Ajouter les \textbf{prix de marché} pour valider l'impact économique
    \item Explorer \textbf{XGBoost} pour prédire le score \textit{avant} cupping
    \item Données de \textbf{fermentation} et de \textbf{torréfaction}
  \end{itemize}
\end{column}
\end{columns}
